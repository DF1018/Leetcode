```python
class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        res = []

        def dfs(i, cur, total):
            if total == target:
                res.append(cur.copy())
                return

            if i >= len(candidates) or total > target:
                return 

# ---------- 選擇 candidates[i] ----------
            cur.append(candidates[i])          # 把目前數字加入組合
            dfs(i, cur, total + candidates[i]) # i 不變，因為同一個數字可以重複選

# ---------- 不選 candidates[i] ----------
            cur.pop()                        # 回溯，把剛剛選的數字拿掉
            dfs(i + 1, cur, total)           # 換下一個數字考慮
        
        dfs(0, [], 0)
        return res
```
